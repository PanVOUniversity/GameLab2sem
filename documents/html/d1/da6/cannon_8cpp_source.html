47 bool Cannon::fire(std::vector<std::vector<Field*>>& board, int targetX, int targetY) {
48  // Cannons use 1 Will point to fire
49  if (getWill() <= 0) {
50  std::cout << getName() << " has no Will to fire.\n";
51  return false;
52  }
53 
54  // Check if the target is within attack distance (using Manhattan distance as per AI logic)
55  // This assumes targetX and targetY are absolute board coordinates, not relative
56  // Need the cannon's current position to calculate distance
57  int cannonX = -1, cannonY = -1;
58  bool foundCannon = false;
59  size_t boardWidth = board.size();
60  size_t boardHeight = board[0].size();
61 
62  // Find the cannon's position
63  for (size_t x = 0; x < boardWidth; ++x) {
64  for (size_t y = 0; y < boardHeight; ++y) {
65  if (board[x][y]->getUnit() == this) {
66  cannonX = static_cast<int>(x);
67  cannonY = static_cast<int>(y);
68  foundCannon = true;
69  break;
70  }
71  }
72  if (foundCannon) break;
73  }
74 
75  if (!foundCannon) {
76  std::cerr << "Error: Cannon unit not found on the board.\n";
77  return false; // Cannon not on board?
78  }
79 
80  int distance = std::abs(targetX - cannonX) + std::abs(targetY - cannonY);
81  if (distance > getAttackDistance()) {
82  std::cout << getName() << " target (" << targetX << "," << targetY << ") is out of range.\n";
83  return false; // Target out of range
84  }
85 
86  // Check if the target field exists and has a unit
87  if (targetX < 0 || static_cast<size_t>(targetX) >= boardWidth || targetY < 0 || static_cast<size_t>(targetY) >= boardHeight) {
88  std::cout << getName() << " target coordinates (" << targetX << "," << targetY << ") are out of bounds.\n";
89  return false; // Target out of bounds
90  }
91 
92  Field* targetField = board[targetX][targetY];
93  if (!targetField || !targetField->getUnit()) {
94  std::cout << getName() << " target field (" << targetX << "," << targetY << ") is empty.\n";
95  return false; // Target field is empty
96  }
97 
98  Unit* target = targetField->getUnit();
99  if (target->getPlayer() == getPlayer()) {
100  std::cout << getName() << " cannot attack a friendly unit.\n";
101  return false; // Cannot attack friendly unit
102  }
103 
104  std::cout << getName() << " fires at " << target->getName() << "!\n";
105 
106  // Apply damage to the target unit
107  target->setHealth(target->getCurrentHealth() - getDamage());
108 
109  // Check if the target unit is defeated
110  if (target->getCurrentHealth() <= 0) {
111  std::cout << target->getName() << " has been defeated!\n";
112  // Remove the defeated unit from the board
113  targetField->setUnit(nullptr);
114  // Note: The actual deletion of the Unit object should be handled by
115  // the game's unit management logic (e.g., the Player class or a central
116  // unit manager) to avoid memory leaks or double deletions.
117  // For simplicity here, we are just removing the pointer from the field.
118  }
119 
120  // Consume Will point for firing
121  setWill(getWill() - 1);
122  std::cout << getName() << " Will remaining: " << getWill() << ".\n";
123 
124  return true; // Fire successful
125 }

189 std::pair<int, int> Cannon::_findEmptyFieldInRange(const std::vector<std::vector<Field*>>& board, int x, int y) const {
190  size_t boardWidth = board.size();
191  size_t boardHeight = board[0].size();
192  int moveRange = getMoveDistance(); // For Cannon, this is 0
193 
194  // Since Cannon moveDistance is 0, it can only "move" to its current empty field if it somehow became empty
195  // This function might not be relevant for a static unit like Cannon unless its moveDistance can change.
196  // However, implementing it for completeness based on the base Unit class.
197  std::vector<std::pair<int, int>> possibleMoves;
198  for (int dx = -moveRange; dx <= moveRange; ++dx) {
199  for (int dy = -moveRange; dy <= moveRange; ++dy) {
200  if (std::abs(dx) + std::abs(dy) > moveRange || (dx == 0 && dy == 0)) {
201  continue;
202  }
203  int newX = x + dx;
204  int newY = y + dy;
205  if (newX >= 0 && static_cast<size_t>(newX) < boardWidth &&
206  newY >= 0 && static_cast<size_t>(newY) < boardHeight)
207  {
208  Field* targetField = board[newX][newY];
209  if(targetField && !targetField->getUnit()) {
210  possibleMoves.push_back({newX, newY});
211  }
212  }
213  }
214  }
215 
216  if (!possibleMoves.empty()) {
217  unsigned seed = std::chrono::steady_clock::now().time_since_epoch().count();
218  std::default_random_engine rng(seed);
219  std::shuffle(possibleMoves.begin(), possibleMoves.end(), rng);
220  return possibleMoves[0];
221  }
222  return {-1, -1}; // No empty field found in range
223 }
