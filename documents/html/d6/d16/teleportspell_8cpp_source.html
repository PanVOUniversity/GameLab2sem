21 bool TeleportSpell::cast(std::vector<std::vector<Field*>>& board, int targetX, int targetY)
22 {
23  // Check if target coordinates are within board bounds
24  size_t boardWidth = board.size();
25  size_t boardHeight = board[0].size();
26  if (targetX < 0 || static_cast<size_t>(targetX) >= boardWidth || targetY < 0 || static_cast<size_t>(targetY) >= boardHeight)
27  {
28  std::cout << "TeleportSpell: Target coordinates out of bounds!\n";
29  return false;
30  }
31 
32  Field* targetField = board[targetX][targetY];
33 
34  // Check if the destination is empty
35  if (targetField->getUnit() != nullptr)
36  {
37  std::cout << "TeleportSpell: Destination (" << targetX << "," << targetY << ") is occupied!\n";
38  return false;
39  }
40 
41  // Find a random friendly unit (assuming the spell is cast by the human player)
42  // A more robust implementation would pass the casting player to the spell.
43  // For now, let's assume the spell is cast by the player whose turn it is and is human.
44  // This requires finding the human player. This might need access to the game state or player list.
45  // Since the spell doesn't have direct access to the Player objects,
46  // the calling code (e.g., getPlayerCommands) needs to handle finding the unit to teleport
47  // and then potentially call a different version of cast or pass the unit.
48 
49  // Given the current structure, let's iterate the board to find a human player unit.
50  // This is not ideal as it doesn't guarantee finding the *intended* unit,
51  // but it fits the cast(board, x, y) signature by using x,y as the destination.
52  Unit* unitToTeleport = nullptr;
53  int unitCurrentX = -1, unitCurrentY = -1;
54  std::vector<std::pair<int, int>> humanUnitCoords;
55 
56  // Find all human player units
57  for (size_t y = 0; y < boardHeight; ++y) {
58  for (size_t x = 0; x < boardWidth; ++x) {
59  Field* currentField = board[x][y];
60  Unit* unit = currentField->getUnit();
61  // Assuming human player units have getPlayer() == true
62  if (unit && unit->getPlayer()) {
63  humanUnitCoords.push_back({static_cast<int>(x), static_cast<int>(y)});
64  }
65  }
66  }
67 
68  if (humanUnitCoords.empty()) {
69  std::cout << "TeleportSpell: No human units found to teleport!\n";
70  return false;
71  }
72 
73  // Pick a random human unit to teleport
74  unsigned seed = std::chrono::steady_clock::now().time_since_epoch().count();
75  std::default_random_engine rng(seed);
76  std::shuffle(humanUnitCoords.begin(), humanUnitCoords.end(), rng);
77 
78  unitCurrentX = humanUnitCoords[0].first;
79  unitCurrentY = humanUnitCoords[0].second;
80  unitToTeleport = board[unitCurrentX][unitCurrentY]->getUnit();
81 
82  if (!unitToTeleport) {
83  // Should not happen if humanUnitCoords was not empty, but as a safeguard
84  std::cerr << "TeleportSpell Error: Selected unit is null.\n";
85  return false;
86  }
87 
88  // Perform the teleportation
89  // Handle onLeave effect for the starting field
90  board[unitCurrentX][unitCurrentY]->onLeave(unitToTeleport);
91 
92  targetField->setUnit(unitToTeleport); // Move the unit to the destination
93  board[unitCurrentX][unitCurrentY]->setUnit(nullptr); // Remove the unit from the source
94 
95  // Handle onEnter effect for the target field
96  targetField->onEnter(unitToTeleport);
97 
98  std::cout << "TeleportSpell: " << unitToTeleport->getName() << " teleported from (" << unitCurrentX << "," << unitCurrentY << ") to (" << targetX << "," << targetY << ").\n";
99 
100  return true; // Spell cast successfully
101 }
