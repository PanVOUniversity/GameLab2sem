47 // Helper function to find a random valid empty adjacent field for building
48 std::pair<int, int> findRandomAdjacentEmptyField(int x, int y, const std::vector<std::vector<Field*>>& board) {
49  size_t boardWidth = board.size();
50  size_t boardHeight = board[0].size();
51  std::vector<std::pair<int, int>> possibleTargets;
52  for(int dx = -1; dx <= 1; ++dx) {
53  for (int dy = -1; dy <= 1; ++dy) {
54  if (dx == 0 && dy == 0) continue;
55  int nx = x + dx;
56  int ny = y + dy;
57  if (nx >= 0 && static_cast<size_t>(nx) < boardWidth &&
58  ny >= 0 && static_cast<size_t>(ny) < boardHeight)
59  {
60  Field* field = board[nx][ny];
61  // Check if field exists, is empty, and not special terrain that blocks building
62  if (field && !field->getUnit() && !dynamic_cast<Hill*>(field) && !dynamic_cast<Fortress*>(field) &&
63  !dynamic_cast<ManaMineField*>(field) && !dynamic_cast<BombedField*>(field) &&
64  !dynamic_cast<ShieldedField*>(field) && !dynamic_cast<BurnedField*>(field) &&
65  !dynamic_cast<HospitalField*>(field)) { // Added HospitalField
66  possibleTargets.push_back({nx, ny});
67  }
68  }
69  }
70  }
71 
72  if (possibleTargets.empty()) {
73  return {-1, -1}; // Indicate no valid target found
74  }
75  unsigned seed = std::chrono::steady_clock::now().time_since_epoch().count();
76  std::default_random_engine rng(seed);
77  std::shuffle(possibleTargets.begin(), possibleTargets.end(), rng);
78  return possibleTargets[0];
79 }

255 bool AIact(Player* aiPlayer, std::vector<std::vector<Field*>>& board) {

263 
264  // Seed the random number generator

266  std::default_random_engine rng(seed);
267 
268  int currentMana = aiPlayer->getMana();
269  int manaSpent = 0; // Declare and initialize manaSpent
270

272  std::uniform_int_distribution<int> dist1(0, currentMana);
273  int manaForUnits = dist1(rng);
274  int manaRemainingAfterUnits = currentMana - manaForUnits;
275

277  int manaForBuild = dist2(rng);
278  int manaRemainingAfterBuild = manaRemainingAfterUnits - manaForBuild;
279 
280  std::uniform_int_distribution<int> dist3(0, manaRemainingAfterBuild);
281  int manaForSpells = dist3(rng);
282  int manaForAttackMove = currentMana - manaForUnits - manaForBuild - manaForSpells; // Mana not allocated to specific actions
283 
284  // --- FIX: Ensure at least one allocation is nonzero if AI has mana ---
285  if (currentMana > 0 && manaForUnits == 0 && manaForBuild == 0 && manaForSpells == 0 && manaForAttackMove == 0) {
286  // Prefer to allocate to units, but fallback to build or spells if needed
287  if (currentMana >= 2) { // Infantry cost
288  manaForUnits = 2;

290  manaForBuild = 3;

627 void tryAIMove(Unit* unit, int x, int y, std::vector<std::vector<Field*>>& board) {

631  size_t boardHeight = board[0].size();

633  std::vector<std::pair<int, int>> safeMoves; // Moves that are not BombedField
634 
635  // Find all valid moves within Manhattan distance
636  for (int dx = -moveRange; dx <= moveRange; ++dx) {
637  for (int dy = -moveRange; dy <= moveRange; ++dy) {
638  if (std::abs(dx) + std::abs(dy) > moveRange || (dx == 0 && dy == 0)) {
639  continue;
640  }

642  int newY = y + dy;

644  newY >= 0 && static_cast<size_t>(newY) < boardHeight)

647  // Check if the target field is a plain field or a hill, and is empty
648  if(targetField && !targetField->getUnit() &&

661  movesToUse = &safeMoves;
662  }
