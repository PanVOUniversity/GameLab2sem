64  size_t boardHeight = board[0].size();
65 
66  // Check the field itself
67  if (unit) {

71  }
72 
73  // Check adjacent fields (including diagonals)
74  for (int dx = -1; dx <= 1; ++dx) {
75  for (int dy = -1; dy <= 1; ++dy) {
76  if (dx == 0 && dy == 0) continue; // Skip the center field
77 
78  int targetX = getX() + dx;
79  int targetY = getY() + dy;
80 
81  if (targetX >= 0 && static_cast<size_t>(targetX) < boardWidth &&
82  targetY >= 0 && static_cast<size_t>(targetY) < boardHeight)
83  {
84  Field* adjacentField = board[targetX][targetY];
85  if (adjacentField && adjacentField->getUnit()) {

87  // applyDamage needs board, x, y, damage, attackingUnit
88  applyDamage(board, targetX, targetY, explosionDamage, nullptr); // Use applyDamage
89  }

91  }
92  }
